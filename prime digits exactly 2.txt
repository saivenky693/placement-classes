public class Main {
    public static void main(String[] args) {
        
        for (int i = 250; i <= 750; i++)
        {
            int n = i;
            int count = 0;
            
            while(n > 0)
            {
                int d = n % 10;
                
                if(d == 2 || d == 3 || d == 5 || d == 7)
                {
                    count++;
                }
                
                n /= 10;
            }
            
            if(count == 2)
            {
                System.out.println(i);
            }
        }
    }
}
